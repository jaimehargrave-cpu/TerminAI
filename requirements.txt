import os
import json

import requests

DEFAULT_MODEL = os.getenv("TERMINAI_MODEL", "llama3.1")
OLLAMA_URL = os.getenv("OLLAMA_URL", "http://localhost:11434")
SYSTEM_PROMPT = (
    "You are TerminAI, a friendly and practical assistant for conversation "
    "and coding help. Explain code clearly, ask clarifying questions when "
    "needed, and never claim you ran code or tests unless you actually did."
)


def call_ollama(prompt, model=DEFAULT_MODEL):
    payload = {
        "model": model,
        "prompt": prompt,
        "stream": False,
        "options": {
            "temperature": 0.7,
        },
    }

    try:
        response = requests.post(
            f"{OLLAMA_URL}/api/generate",
            json=payload,
            timeout=120,
        )
        response.raise_for_status()
    except requests.RequestException as exc:
        raise RuntimeError(f"Failed to reach Ollama at {OLLAMA_URL}: {exc}") from exc

    try:
        data = response.json()
    except ValueError as exc:
        raise RuntimeError("Ollama returned invalid JSON.") from exc

    if "response" not in data:
        raise RuntimeError(f"Unexpected Ollama response: {data}")

    return data["response"].strip()


def build_prompt(history):
    messages = [{"role": "system", "content": SYSTEM_PROMPT}]
    for item in history:
        messages.append({"role": item["role"], "content": item["content"]})
    return json.dumps(messages)


def main():
    print(f"TerminAI — local AI via Ollama ({DEFAULT_MODEL})")
    print(f"Ollama server: {OLLAMA_URL}")
    print("Type /clear to reset the conversation or /exit to quit.\n")

    history = []

    while True:
        try:
            message = input("You: ").strip()
        except (KeyboardInterrupt, EOFError):
            print("\nTerminAI: Goodbye!")
            break

        if not message:
            continue

        if message.lower() in {"/exit", "/quit"}:
            print("TerminAI: Goodbye!")
            break

        if message.lower() == "/clear":
            history = []
            print("TerminAI: Conversation cleared.\n")
            continue

        history.append({"role": "user", "content": message})

        try:
            prompt = build_prompt(history)
            answer = call_ollama(prompt, model=DEFAULT_MODEL)
            print(f"\nTerminAI: {answer}\n")
            history.append({"role": "assistant", "content": answer})
        except RuntimeError as exc:
            history.pop()
            print(f"\nTerminAI: {exc}\n")


if __name__ == "__main__":
    main()
